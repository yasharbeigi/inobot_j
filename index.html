آشنایی با اینوبات و اولین برنامه


این جزوه مسیر یادگیری برنامه‌نویسی اینوبات را از اولین برنامه تا ساخت پروژه‌های ترکیبی دنبال می‌کند.
هدف این است که هنرجو فقط نام دستورها را حفظ نکند؛ بلکه بداند هر دستور چه ورودی‌ای می‌گیرد، چه خروجی‌ای دارد،
در چه زمانی باید از آن استفاده کند و چگونه چند قابلیت را در یک برنامه ترکیب کند.

الگوی پایه برنامه:
1. افزودن فایل اصلی.
2. ساخت شیء ربات.
3. راه‌اندازی در setup.
4. اجرای منطق برنامه در loop.

در تمام مثال‌های این جزوه کد به صورت Sketch کامل ارائه شده تا بتوان آن را یک‌جا در محیط برنامه‌نویسی وارد کرد.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();
}

void loop() {
}

LED؛ اولین خروجی دیجیتال


LED یکی از ساده‌ترین خروجی‌ها برای شروع است. تابع setLED دو ورودی دارد: پایه LED و وضعیت آن.
HIGH برای روشن و LOW برای خاموش‌کردن استفاده می‌شود.

نکته آموزشی: delay اجرای برنامه را برای مدت مشخصی متوقف می‌کند؛ بنابراین برای چشمک‌زدن ساده مناسب است.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();
}

void loop() {
  bot.setLED(PIN_LED_1, HIGH);
  delay(500);

  bot.setLED(PIN_LED_1, LOW);
  delay(500);
}

کلیدها؛ تبدیل ورودی به تصمیم


تابع readBTN وضعیت یک کلید را می‌خواند. با ترکیب آن با if می‌توان برای ربات تصمیم ساخت.
در مثال زیر با فشار کلید اول، LED اول روشن می‌شود و با رهاکردن خاموش می‌شود.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();
}

void loop() {
  uint8_t state = bot.readBTN(PIN_BTN_1);

  if (state == HIGH) {
    bot.setLED(PIN_LED_1, HIGH);
  } else {
    bot.setLED(PIN_LED_1, LOW);
  }

  delay(20);
}

سروو؛ کنترل زاویه


سروو با setServoPositions کنترل می‌شود. شیء سروو باید یکی از servo_1 تا servo_8 باشد.
مقدار position در برنامه به صورت uint8_t دریافت می‌شود و برای زاویه‌های معمول سروو، بازه 0 تا 180 کاربرد دارد.

#include <RobotixInnoBot.h>

int i = 0;

RobotixInnoBot bot;

void setup() {
  bot.begin();
}

void loop() {
  for (i = 0; i <= 180; i++) {
    bot.setServoPositions(bot.servo_1, i);
    delay(15);
  }

  for (i = 180; i >= 0; i--) {
    bot.setServoPositions(bot.servo_1, i);
    delay(15);
  }
}

موتور؛ کنترل یک موتور


setMotorSpeed یک موتور و سرعت آن را دریافت می‌کند. سرعت مثبت و منفی جهت را تغییر می‌دهند و صفر باعث توقف موتور می‌شود.
محدوده معمول سرعت در این برنامه بر پایه signed 32-bit است، اما در پیاده‌سازی PWM مقدار مورد استفاده عملاً برای ورودی‌های 0 تا 255 طراحی شده است.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();
}

void loop() {
  bot.setMotorSpeed(bot.motor_1, 180);
  delay(1500);

  bot.setMotorSpeed(bot.motor_1, 0);
  delay(800);

  bot.setMotorSpeed(bot.motor_1, -180);
  delay(1500);

  bot.setMotorSpeed(bot.motor_1, 0);
  delay(800);
}

  bot.speed(-150, 150);
  delay(800);

  bot.speed(0, 0);
  delay(700);

  bot.speed(-150, -150);
  delay(2000);

  bot.speed(0, 0);

RGB؛ ساخت نوار رنگی


می‌توان با یک حلقه، چندین پیکسل را پشت سر هم روشن کرد. در این برد آرایه RGB با 16 پیکسل ساخته می‌شود.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();
  bot.rjRgbInit(bot.port_1);
}

void loop() {
  for (int pixel = 0; pixel < 16; pixel++) {
    bot.rjRgbWrite(bot.port_1, pixel, 40, 120, 255);
    delay(50);
  }

  delay(500);
  bot.rjRgbClear(bot.port_1);
  delay(500);
}

تشخیص رنگ


rjTcsRead شش خروجی دارد: R، G، B، C، دمای رنگ و لوکس. این داده‌ها برای نمایش و تحلیل رنگ مناسب‌اند.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  Serial.begin(115200);
  bot.begin();

  bot.rjTcsInit(bot.port_1);
}

void loop() {
  uint16_t r, g, b, c;
  uint16_t colorTemp, lux;

  bot.rjTcsRead(
    bot.port_1,
    r, g, b, c,
    colorTemp, lux
  );

  Serial.print("R: "); Serial.println(r);
  Serial.print("G: "); Serial.println(g);
  Serial.print("B: "); Serial.println(b);
  Serial.print("C: "); Serial.println(c);
  Serial.print("Color temperature: "); Serial.println(colorTemp);
  Serial.print("Lux: "); Serial.println(lux);

  delay(700);
}

آرایه چهارگانه IR


rjIrArrayRead یک آرایه چهارخانه را پر می‌کند. هر خانه یک byte است و می‌توان چهار وضعیت را در یک حلقه چاپ یا تحلیل کرد.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  Serial.begin(115200);
  bot.begin();

  bot.rjIrArrayInit(bot.port_1);
}

void loop() {
  uint8_t sensors[4] = {0, 0, 0, 0};

  bot.rjIrArrayRead(bot.port_1, sensors);

  for (int i = 0; i < 4; i++) {
    Serial.print("Sensor ");
    Serial.print(i);
    Serial.print(": ");
    Serial.println(sensors[i]);
  }

  Serial.println("----------------");
  delay(200);
}

در این پروژه سه مفهوم ورودی، خروجی نوری و خروجی صوتی ترکیب می‌شوند.
با فشار کلید، چراغ و بیزر فعال می‌شوند.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();

  bot.setLED(PIN_LED_1, LOW);
  bot.buzzerNoTone();
}

void loop() {
  if (bot.readBTN(PIN_BTN_1) == HIGH) {
    bot.setLED(PIN_LED_1, HIGH);
    bot.buzzerTone(1000);
  } else {
    bot.setLED(PIN_LED_1, LOW);
    bot.buzzerNoTone();
  }

  delay(20);
}

پروژه کامل: هشدار فاصله


سنسور فاصله و بیزر در یک حلقه ترکیب شده‌اند. اگر فاصله کمتر از 20 سانتی‌متر باشد، هشدار فعال می‌شود.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();
  bot.rjSrfInit(bot.port_1);
}

void loop() {
  float distance = bot.rjSrfRead(bot.port_1);

  if (distance > 0 && distance < 20) {
    bot.setLED(PIN_LED_1, HIGH);
    bot.buzzerTone(1400);
  } else {
    bot.setLED(PIN_LED_1, LOW);
    bot.buzzerNoTone();
  }

  delay(50);
}

پروژه کامل: حرکت و توقف بر اساس فاصله


این نمونه یک الگوی پایه برای ربات‌های خودکار است: سنسور تصمیم می‌گیرد و موتور اجرا می‌کند.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();
  bot.rjSrfInit(bot.port_1);
}

void loop() {
  float distance = bot.rjSrfRead(bot.port_1);

  if (distance > 20) {
    bot.speed(150, 150);
  } else {
    bot.speed(0, 0);
    bot.buzzerTone(1200);
    delay(250);
    bot.buzzerNoTone();
  }

  delay(50);
}

پروژه کامل: نمایش وضعیت سنسور و RGB


در این پروژه مقدار سنسور IR خوانده و به صورت شدت تقریبی روی یک پیکسل RGB نمایش داده می‌شود.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  Serial.begin(115200);
  bot.begin();

  bot.rjIrInit(bot.port_1);
  bot.rjRgbInit(bot.port_2);
}

void loop() {
  int value = bot.rjIrRead(bot.port_1);

  Serial.print("IR: ");
  Serial.println(value);

  if (value < 300) {
    bot.rjRgbWrite(bot.port_2, 0, 0, 255, 0);
  } else if (value < 700) {
    bot.rjRgbWrite(bot.port_2, 0, 255, 120, 0);
  } else {
    bot.rjRgbWrite(bot.port_2, 0, 255, 0, 0);
  }

  delay(100);
}

پروژه کامل: حرکت سروو به کمک سنسور فاصله


وقتی مانع نزدیک شود، سروو به موقعیت 180 می‌رود؛ در غیر این صورت به 0 برمی‌گردد.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  bot.begin();
  bot.rjSrfInit(bot.port_1);
  bot.setServoPositions(bot.servo_1, 0);
}

void loop() {
  float distance = bot.rjSrfRead(bot.port_1);

  if (distance > 0 && distance < 20) {
    bot.setServoPositions(bot.servo_1, 180);
  } else {
    bot.setServoPositions(bot.servo_1, 0);
  }

  delay(100);
}

مرجع پورت‌ها و اشیای عمومی


دوازده پورت عمومی به صورت port_1 تا port_12 در دسترس‌اند. چهار موتور motor_1 تا motor_4 و هشت سروو servo_1 تا servo_8 نیز به صورت عضو عمومی شیء bot در اختیار برنامه هستند.

برای کار با یک ماژول RJ، ابتدا پورت مناسب را انتخاب می‌کنیم، سپس Init همان ماژول را روی آن اجرا می‌کنیم و در loop مقدار را می‌خوانیم.

#include <RobotixInnoBot.h>

RobotixInnoBot bot;

void setup() {
  Serial.begin(115200);
  bot.begin();

  bot.rjSrfInit(bot.port_1);
}

void loop() {
  float distance = bot.rjSrfRead(bot.port_1);

  Serial.print("Port 1 distance: ");
  Serial.print(distance);
  Serial.println(" cm");

  delay(100);
}

نکات حرفه‌ای و خطاهای رایج


۱) قبل از استفاده از ماژول RJ، Init مناسب همان ماژول را اجرا کنید.
۲) پورت انتخاب‌شده باید با سیم‌کشی واقعی مطابقت داشته باشد.
۳) Serial.begin را فقط یک بار در setup قرار دهید.
۴) در برنامه‌های موتور، ابتدا سرعت کم را آزمایش کنید.
۵) برای سروو، محدوده مکانیکی واقعی سازوکار را در نظر بگیرید.
۶) delayهای بسیار بزرگ باعث کندشدن واکنش ربات می‌شوند.
۷) وقتی مقدار سنسور غیرعادی است، ابتدا مقدار خام را در Serial Monitor ببینید.
۸) در پروژه‌های ترکیبی، هر بخش را جداگانه آزمایش و سپس ترکیب کنید.
